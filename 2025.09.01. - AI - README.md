            const float ASXAIController::PatrolRepeatInterval(3.f);

            void ASXAIController::BeginPlay()
            {
            	Super::BeginPlay();
            
            	GetWorld()->GetTimerManager().SetTimer(PatrolTimerHandle, this, &ThisClass::OnPatrolTimerElapsed, PatrolRepeatInterval, true);
            }
            
            void ASXAIController::EndPlay(const EEndPlayReason::Type EndPlayReason)
            {
            	GetWorld()->GetTimerManager().ClearTimer(PatrolTimerHandle);
            
            	Super::EndPlay(EndPlayReason);
            }
            
            void ASXAIController::OnPatrolTimerElapsed()
            {
            	APawn* ControlledPawn = GetPawn();
            	if (IsValid(ControlledPawn) == true)
            	{
            		UNavigationSystemV1* NavigationSystem = UNavigationSystemV1::GetNavigationSystem(GetWorld());
            		if (IsValid(NavigationSystem) == true)
            		{
            			FVector ActorLocation = ControlledPawn->GetActorLocation();
            			FNavLocation NextLocation;
            			if (NavigationSystem->GetRandomPointInNavigableRadius(ActorLocation, PatrolRadius, NextLocation) == true)
            			{
            				UAIBlueprintHelperLibrary::SimpleMoveToLocation(this, NextLocation.Location);
            			}
            		}
            	}
            }

    - 8.2 Behavior Tree and Blackboard
        - Finite State Machine
        상태 노드들을 연결하여 인공지능을 설계하는 기법.
        모든 상태 노드들은 상하 관계가 없음. 수평 구조.
        즉, 모든 상태 노드에서 모든 상태 노드로 전환이 가능함. 구현하기가 쉬움.
        이런 자유로움 때문에 상태 노드가 많아질수록 굉장히 복잡해짐.
        
        - Behavior Tree
        우선 순위와 트리 구조를 사용해서 인공지능을 설계하는 기법.
        즉, 상하 관계가 있음. 수직 구조.
        상위 상태 노드들은 어떤 서브 상태 트리를 동작시킬지 결정함.
        하위 상태 노드들은 자신이 맡은 작업만 수행하는 구조.
        적절한 규칙을 통해서 노드가 많아져도 가독성이 높다는 장점이 있음.
        규칙과 다양한 노드들의 용도를 알아야해서 구현하기가 까다로울 수 있음.
        
        - Blackboard
        Behavior Tree가 판단하는데 필요한 데이터들을 저장한 애셋.
        매순간 업데이트되며 Behavior Tree의 동작에 영향을 줌.
        
        - Behavior Tree의 규칙
        트리의 위, 좌측일수록 우선순위가 높음.
        시작 노드를 설정할 필요 없음. 항상 왼쪽 노드부터 깊이 우선 탐색 진행.
        각 노드를 Task라고 부름.
       
        
        - 비헤이비어 트리는 상위 노드에서 어떤 하위 서브 트리를 실행할지 결정함
        이런 상위 노드를 Composite Task라고 부름.
        Composite Task는 아래와 같은 종류가 있음. 자동 라면을 끓여주는 기계를 예시로 듦.
        ***Sequence***: 여러 Child Task를 왼쪽부터 순서대로 수행
        ex) 1번 Child Task 라면 꺼내기. 2번 Child Task 냄비에 물담기. 3번 Child Task 물끓이기
        ***Selector***: 여러 Child Task 중 조건에 맞는 하나의 Child Task를 선택해서 수행
        ex) 주문에 따라 1번 Child Task 신라면 꺼내기. 2번 Child Task 너구리 꺼내기. 중 하나
        ***Parellel***: 여러 Child Task를 동시에 수행
        ex) 1번 Child Task 물 끓이기. 2번 Child Task 라면 내용물을 냄비에 넣는다. 를 동시에.
       
        
        - Task 수행 결과
        InProgress: Task 수행 중. 수행 결과는 나중에 알려줄 예정이다.
        Succeeded: Task 수행 성공.
        Failed: 내부 요인으로 인한 Task 수행 실패. 수행은 했지만 실패함.
        Aborted: 외부 요인으로 인한 Task 수행 실패. 수행 중에 중단됨.
        - Child Task 수행 결과에 따른 Composite Task의 반응
        ***Sequence***: 실패한 Child Task가 나올 때까지 수행.
          ex) 라면이 없으면 냄비에 물을 담지도 않음. 물이 안나오면 물을 끓지도 않음.
        ***Selector***: 실패한 Child Task는 넘어가고 다음 Child Task 수행.
          ex) 주문에 따라 신라면을 골랐지만, 신라면이 없다면 너구리를 꺼냄.
        ***Paraller***: 실패한 Child Task가 나올 때까지 수행.
          ex) 물이 끓일 수 없다면 라면의 내용물을 냄비에 넣는 것도 중지.
        
        - Composite Task에 부착 가능한 기능.
        ***Decorator***: 해당 Composite Task가 실행되는 조건을 지정.
        ex) 주문이 만약 신라면이라면, 주문이 만약 너구리라면.
        ***Service***: 해당 Composite Task가 수행중일때 Tick() 함수가 호출되게끔 함.
        ex) 물이 끓는지 일정 시간마다 확인하고자 할 때.
        ***Observer Abort***: Decorator 조건에 부합되면 해당 Composite Task 내 작업 모두 중지.
        ex) 모종의 이유로 라면 주문이 취소되었다면, 
          현재 수행 중인 Composite Task 내 모든 Child Task 중지. 루트 노드부터 다시 수행.

    - 8.6 Decorator
        - Decorator
        Blackboard의 값을 기반으로 특정 Composite Task의 수행 분기를 결정하는 노드.
        
        - Decorator에 따라 추격과 순찰로 나누기
            
            BT_NPC > 왼쪽 Sequence Task 우클릭 > Add Decorator > Blackboard 클릭
            해당 Decorator 클릭 > Details > Notify Observer를 On Value Change로 설정.
            Observer Aborts를 Self로 설정
            오른쪽 Sequence Task 우클릭 > Add Decorator > Blackboard 클릭.
            왼쪽의 반대 조건인 Is Not Set을 지정.
            
           <img width="346" height="725" alt="스크린샷 2025-09-01 182039" src="https://github.com/user-attachments/assets/b4a046c1-729f-4121-b360-91a3c9fa38e8" />

        - Notify Observer
        On Value Change는 해당 키 값의 변경이 감지되면, Notify Observer가 발생함.
        On Result Change는 결과가 변경될 때 Notify Observer가 발생함.
        ex) 1.f < x 라는 조건식이 있을때, x가 0.8f에서 0.9f가 되면 On Value Change만 발생.
          x가 1.f에서 1.1f가 되면 On Value Change와 On Result Change가 둘 다 발생.
        
        - Observer Aborts
        Notify Observer가 발생했을 때 어떤 서브 트리를 중단(Abort)할지 결정하는 옵션.
        None으로 두면, 해당 Composite Task의 모든 자식 Task들이 
        마무리 될때까지 중단되지 않음. 즉, 아무것도 중단 안함.
        Self로 두면 해당 Composite Task의 자식 Task들의 수행을 즉시 중단.
        Lower Priority는 자신의 우측 노드들을 모두 중단.
        Both는 Self + Lower Priority. 하위 서브트리와 오른쪽 모든 노드를 중단함.

            EBTNodeResult::Type UBTTask_TurnToTarget::ExecuteTask(UBehaviorTreeComponent& OwnerComp, uint8* NodeMemory)
            {
            	EBTNodeResult::Type Result = Super::ExecuteTask(OwnerComp, NodeMemory);
            
            	ASXAIController* AIController = Cast<ASXAIController>(OwnerComp.GetAIOwner());
            	checkf(IsValid(AIController) == true, TEXT("Invalid AIController."));
            
            	ASXNonPlayerCharacter* NPC = Cast<ASXNonPlayerCharacter>(AIController->GetPawn());
            	checkf(IsValid(NPC) == true, TEXT("Invalid NPC."));
            
            	if (ASXCharacterBase* TargetPC = Cast<ASXCharacterBase>(OwnerComp.GetBlackboardComponent()->GetValueAsObject(AIController->TargetCharacterKey)))
            	{
            		FVector LookVector = TargetPC->GetActorLocation() - NPC->GetActorLocation();
            		LookVector.Z = 0.f;
            		FRotator TargetRotation = FRotationMatrix::MakeFromX(LookVector).Rotator();
            		NPC->SetActorRotation(FMath::RInterpTo(NPC->GetActorRotation(), TargetRotation, GetWorld()->GetDeltaSeconds(), 2.f));
            
            		return Result = EBTNodeResult::Succeeded;
            	}
            
            	return Result = EBTNodeResult::Failed;
            }
