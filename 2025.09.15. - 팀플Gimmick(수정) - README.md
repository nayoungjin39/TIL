```c++ TCharacter.cpp
// 서버에서 바인드 스킬 발동
void ATCharacter::Server_UseSkill_Implementation()
{
    if (!bCanUseBindSkill) return;

    ATPlayerState* PS = GetPlayerState<ATPlayerState>();
    if (PS && PS->Team == ETeam::Thief) // 도둑만 가능
    {
        if (BindSkillActorClass)
        {
            bCanUseBindSkill = false;

            GetWorld()->SpawnActor<ATBind>(BindSkillActorClass, GetActorLocation(), GetActorRotation());
        }
    }
}

// 스피드 버프 적용 & 해제
void ATCharacter::ApplySpeedBuff(float Multiplier, float Duration)
{
    if (HasAuthority())
    {
        SpeedBuffMultiplier = Multiplier;
        OnRep_SpeedBuffMultiplier();
        GetWorld()->GetTimerManager().SetTimer(SpeedBuffTimerHandle, this, &ATCharacter::EndSpeedBuff, Duration, false);
    }
}

void ATCharacter::EndSpeedBuff()
{
    if (HasAuthority())
    {
        SpeedBuffMultiplier = 1.0f;
        OnRep_SpeedBuffMultiplier();
    }
}

void ATCharacter::OnRep_SpeedBuffMultiplier()
{
    UpdateMovementSpeed();
}
```

## 3. 바인드 스킬 액터 (경찰 묶기)

```c++ TBind.cpp
// 오버랩 발생 시 (경찰만 적용)
void ATBind::OnSphereOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor, ...)
{
    ACharacter* Character = Cast<ACharacter>(OtherActor);
    if (Character && Character->ActorHasTag(FName("Tagger")))
    {
        ApplyBindEffect(Character);
    }
}

void ATBind::ApplyBindEffect(ACharacter* Character)
{
    UCharacterMovementComponent* MoveComp = Character->GetCharacterMovement();
    if (MoveComp)
    {
        MoveComp->DisableMovement(); // 이동 불가 처리
        ActiveBoundCharacters.Add(Character);
    }
}

// 일정 시간 후 해제
void ATBind::OnDisappear()
{
    for (auto WeakChar : ActiveBoundCharacters)
    {
        if (ACharacter* BoundChar = WeakChar.Get())
        {
            RemoveBindEffect(BoundChar);
        }
    }
    Destroy();
}
```

```c++ TSpeedup.cpp
// 오버랩 발생 시 (도둑만 효과)
void ATSpeedup::OnSphereOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor, ...)
{
    ATCharacter* TChar = Cast<ATCharacter>(OtherActor);
    if (TChar && TChar->ActorHasTag(FName("Hider"))) // 도둑만 가능
    {
        TChar->ApplySpeedBuff(SpeedMultiplier, BuffDuration);
        Destroy(); // 한 번 쓰면 사라짐
    }
}
```
